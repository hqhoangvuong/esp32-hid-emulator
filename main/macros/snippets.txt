# Pools for developer.txt. POOL name ... END holds snippets separated by lines that are exactly
# ---; POOL name LINES makes every non-blank line one item. Bodies are literal: '#' lines and
# blank lines inside a POOL are content, and the body ends at a line that is exactly END.
#
# Snippet grammar (enforced by tools/check_script), so VS Code's auto-closed brackets land
# where the text says:
#   - no leading or trailing whitespace, no tabs, printable ASCII, never ///
#   - a line is exactly } or };, or is balanced, or is balanced up to a final {
#   - no blank line and no { line directly before a } line (write { } on one line instead)
#   - so no "} else {" or "} catch {": put the closing brace on its own line
# Types may use only the implicit usings (System, System.Linq, System.Collections.Generic,
# System.IO, System.Threading, System.Threading.Tasks).

public class RetryPolicy {
public int MaxAttempts { get; init; } = 3;
public async Task<T> RunAsync<T>(Func<Task<T>> action) {
for (var attempt = 1; ; attempt++) {
try {
return await action();
}
catch (Exception) when (attempt < MaxAttempts) {
await Task.Delay(100 * attempt);
}
}
}
}

public class LruCache<TKey, TValue> where TKey : notnull {
private readonly int _capacity;
private readonly Dictionary<TKey, LinkedListNode<(TKey Key, TValue Value)>> _map = new();
private readonly LinkedList<(TKey Key, TValue Value)> _order = new();
public LruCache(int capacity) => _capacity = capacity;
public bool TryGet(TKey key, out TValue value) {
if (_map.TryGetValue(key, out var node)) {
_order.Remove(node);
_order.AddFirst(node);
value = node.Value.Value;
return true;
}
value = default!;
return false;
}
public void Put(TKey key, TValue value) {
if (_map.Remove(key, out var old)) {
_order.Remove(old);
}
else if (_map.Count >= _capacity) {
_map.Remove(_order.Last!.Value.Key);
_order.RemoveLast();
}
_map[key] = _order.AddFirst((key, value));
}
}

public static class ChunkExtensions {
public static IEnumerable<List<T>> InChunks<T>(this IEnumerable<T> source, int size) {
var chunk = new List<T>(size);
foreach (var item in source) {
chunk.Add(item);
if (chunk.Count == size) {
yield return chunk;
chunk = new List<T>(size);
}
}
if (chunk.Count > 0) {
yield return chunk;
}
}
}

public class TokenBucket {
private readonly int _capacity;
private double _tokens;
private DateTime _last = DateTime.UtcNow;
public TokenBucket(int capacity) {
_capacity = capacity;
_tokens = capacity;
}
public bool TryTake(double ratePerSecond) {
var now = DateTime.UtcNow;
_tokens = Math.Min(_capacity, _tokens + (now - _last).TotalSeconds * ratePerSecond);
_last = now;
if (_tokens < 1) {
return false;
}
_tokens -= 1;
return true;
}
}

public static class RangeParser {
public static (int Start, int End) Parse(string text) {
var dash = text.IndexOf('-');
if (dash < 0) {
var single = int.Parse(text);
return (single, single);
}
return (int.Parse(text[..dash]), int.Parse(text[(dash + 1)..]));
}
}

public class EventBus {
private readonly Dictionary<Type, List<Action<object>>> _handlers = new();
public void Subscribe<T>(Action<T> handler) {
if (!_handlers.TryGetValue(typeof(T), out var list)) {
list = new List<Action<object>>();
_handlers[typeof(T)] = list;
}
list.Add(e => handler((T)e));
}
public void Publish<T>(T message) where T : notnull {
if (_handlers.TryGetValue(typeof(T), out var list)) {
foreach (var handler in list) {
handler(message);
}
}
}
}

public class Page<T> {
public IReadOnlyList<T> Items { get; }
public int Number { get; }
public int TotalPages { get; }
public Page(IReadOnlyList<T> all, int number, int size) {
Number = number;
TotalPages = (all.Count + size - 1) / size;
Items = all.Skip((number - 1) * size).Take(size).ToList();
}
public bool HasNext => Number < TotalPages;
}

public sealed class Debouncer {
private readonly TimeSpan _wait;
private CancellationTokenSource? _cts;
public Debouncer(TimeSpan wait) => _wait = wait;
public async Task RunAsync(Func<Task> action) {
_cts?.Cancel();
_cts = new CancellationTokenSource();
var token = _cts.Token;
try {
await Task.Delay(_wait, token);
await action();
}
catch (TaskCanceledException) {
return;
}
}
}

public static class WordStats {
public static Dictionary<string, int> Count(string text) {
var words = text.ToLowerInvariant().Split(' ', StringSplitOptions.RemoveEmptyEntries);
return words.GroupBy(w => w).ToDictionary(g => g.Key, g => g.Count());
}
public static string Longest(string text) {
return text.Split(' ').OrderByDescending(w => w.Length).FirstOrDefault() ?? "";
}
}

public class MovingAverage {
private readonly Queue<double> _window = new();
private readonly int _size;
private double _sum;
public MovingAverage(int size) => _size = size;
public double Add(double value) {
_window.Enqueue(value);
_sum += value;
if (_window.Count > _size) {
_sum -= _window.Dequeue();
}
return _sum / _window.Count;
}
}

public readonly record struct SemVer(int Major, int Minor, int Patch) : IComparable<SemVer> {
public static SemVer Parse(string text) {
var p = text.Split('.').Select(int.Parse).ToArray();
return new SemVer(p[0], p[1], p[2]);
}
public int CompareTo(SemVer other) {
var major = Major.CompareTo(other.Major);
if (major != 0) {
return major;
}
var minor = Minor.CompareTo(other.Minor);
return minor != 0 ? minor : Patch.CompareTo(other.Patch);
}
}

public class Inventory {
private readonly Dictionary<string, int> _stock = new();
public void Add(string sku, int count) {
_stock[sku] = _stock.GetValueOrDefault(sku) + count;
}
public bool TryRemove(string sku, int count) {
if (_stock.GetValueOrDefault(sku) < count) {
return false;
}
_stock[sku] -= count;
return true;
}
public int Total => _stock.Values.Sum();
}
